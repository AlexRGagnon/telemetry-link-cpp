cmake_minimum_required(VERSION 3.20)
project(telemetry_link VERSION 0.1.0 LANGUAGES CXX)
option(TELEMETRY_SANITIZERS "Enable ASan and UBSan with GCC/Clang" OFF)
add_library(telemetry_link src/protocol.cpp)
target_include_directories(telemetry_link PUBLIC include)
target_compile_features(telemetry_link PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(telemetry_link PRIVATE /W4 /WX)
else()
  target_compile_options(telemetry_link PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(TELEMETRY_SANITIZERS)
    target_compile_options(telemetry_link PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(telemetry_link PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(telemetry-demo src/main.cpp)
target_link_libraries(telemetry-demo PRIVATE telemetry_link)
add_executable(telemetry-tests tests/protocol_tests.cpp)
target_link_libraries(telemetry-tests PRIVATE telemetry_link)
enable_testing()
foreach(case crc cobs roundtrip limits errors stream oversize recovery sequence fuzz)
  add_test(NAME ${case} COMMAND telemetry-tests ${case})
endforeach()
